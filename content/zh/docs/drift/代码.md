#include <stdio.h> 

void mulSum(int n);
int main(){
    printf("Enter an num\n");
    mulSum(5);

    return 0;
}

void mulSum(int n){
    int s, i, mul;
    for(i = 1, s = 0, mul = 1; i <= n; i++){
        mul *= i;
        s += mul;
    }
    printf("%d\n", s);

}
